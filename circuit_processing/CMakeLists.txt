cmake_minimum_required(VERSION 3.16)
project(fixed_roi_validation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
file(GLOB UNIT_SOURCES CONFIGURE_DEPENDS third_party/evoapproxlib/add12se_*.c)
add_library(evoapprox_units STATIC ${UNIT_SOURCES} src/approximate/evoapprox_adapter.cpp)
target_include_directories(evoapprox_units PUBLIC include ${OpenCV_INCLUDE_DIRS})
add_executable(fixed_roi_verify app/fixed_roi_verify.cpp)
target_link_libraries(fixed_roi_verify PRIVATE evoapprox_units ${OpenCV_LIBS})
